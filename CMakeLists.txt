cmake_minimum_required(VERSION 3.24)
project(quantforge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(OpenMP REQUIRED)
add_library(qf_cpu SHARED native/cpu.cpp)
target_link_libraries(qf_cpu PRIVATE OpenMP::OpenMP_CXX)
add_library(qf_research SHARED native/research.cpp)
target_link_libraries(qf_research PRIVATE OpenMP::OpenMP_CXX)
option(QF_CUDA "Build optional CUDA engine" OFF)
if(QF_CUDA)
  enable_language(CUDA)
  add_library(qf_gpu SHARED native/gpu.cu)
  set_target_properties(qf_gpu PROPERTIES CUDA_STANDARD 20)
endif()
